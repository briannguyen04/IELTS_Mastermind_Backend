USE ielts_mastermind_db;

SET @practice_content_id = '2cd968df-4e59-4b30-85e3-4b2cee6aaf8a';

START TRANSACTION;

DROP TEMPORARY TABLE IF EXISTS tmp_old_practice_questions;
DROP TEMPORARY TABLE IF EXISTS tmp_practice_questions;

CREATE TEMPORARY TABLE tmp_old_practice_questions AS
SELECT practice_question_id
FROM practice_question
WHERE practice_content_id = @practice_content_id;

DELETE FROM practice_question_answer
WHERE practice_question_id IN (
    SELECT practice_question_id
    FROM tmp_old_practice_questions
);

DELETE FROM practice_question
WHERE practice_question_id IN (
    SELECT practice_question_id
    FROM tmp_old_practice_questions
);

CREATE TEMPORARY TABLE tmp_practice_questions AS
WITH RECURSIVE seq(n) AS (
    SELECT 1
    UNION ALL
    SELECT n + 1
    FROM seq
    WHERE n < 40
)
SELECT
    UUID() AS practice_question_id,
    n AS order_index,
    CASE
        WHEN n BETWEEN 1 AND 6 THEN 'MATCHING_INFORMATION'
        WHEN n BETWEEN 7 AND 13 THEN 'SUMMARY_COMPLETION'
        WHEN n BETWEEN 14 AND 17 THEN 'MATCHING_INFORMATION'
        WHEN n BETWEEN 18 AND 22 THEN 'SUMMARY_COMPLETION'
        WHEN n BETWEEN 23 AND 26 THEN 'MATCHING_FEATURES'
        WHEN n BETWEEN 27 AND 31 THEN 'MULTIPLE_CHOICE'
        WHEN n BETWEEN 32 AND 36 THEN 'SUMMARY_COMPLETION'
        WHEN n BETWEEN 37 AND 40 THEN 'IDENTIFYING_WRITERS_VIEWS_CLAIMS_YES_NO_NOT_GIVEN'
    END AS type,
    CASE
        WHEN n BETWEEN 1 AND 13 THEN 'HOUSING_CITIES_AND_URBAN_RURAL_LIFE'
        WHEN n BETWEEN 14 AND 26 THEN 'ANIMALS_AND_WILDLIFE'
        WHEN n BETWEEN 27 AND 40 THEN 'SOCIETY_SOCIAL_BEHAVIOR_AND_VALUES'
    END AS topic_tag,
    @practice_content_id AS practice_content_id
FROM seq;

INSERT INTO practice_question (
    practice_question_id,
    order_index,
    type,
    topic_tag,
    practice_content_id
)
SELECT
    practice_question_id,
    order_index,
    type,
    topic_tag,
    practice_content_id
FROM tmp_practice_questions;

INSERT INTO practice_question_answer (
    practice_question_id,
    answer_index,
    answer_value
)
SELECT
    q.practice_question_id,
    a.answer_index,
    a.answer_value
FROM tmp_practice_questions q
JOIN (
    SELECT 1 AS order_index, 0 AS answer_index, 'B' AS answer_value
    UNION ALL SELECT 2, 0, 'C'
    UNION ALL SELECT 3, 0, 'F'
    UNION ALL SELECT 4, 0, 'D'
    UNION ALL SELECT 5, 0, 'E'
    UNION ALL SELECT 6, 0, 'A'
    UNION ALL SELECT 7, 0, 'safety'
    UNION ALL SELECT 8, 0, 'traffic'
    UNION ALL SELECT 9, 0, 'carriageway'
    UNION ALL SELECT 10, 0, 'mobile'
    UNION ALL SELECT 11, 0, 'dangerous'
    UNION ALL SELECT 12, 0, 'communities'
    UNION ALL SELECT 13, 0, 'healthy'
    UNION ALL SELECT 14, 0, 'F'
    UNION ALL SELECT 15, 0, 'A'
    UNION ALL SELECT 16, 0, 'D'
    UNION ALL SELECT 17, 0, 'A'
    UNION ALL SELECT 18, 0, 'genetic traits'
    UNION ALL SELECT 19, 0, 'heat loss'
    UNION ALL SELECT 20, 0, 'ears'
    UNION ALL SELECT 21, 0, 'fat'
    UNION ALL SELECT 21, 1, 'insulating fat'
    UNION ALL SELECT 22, 0, 'emissions'
    UNION ALL SELECT 22, 1, 'carbon emissions'
    UNION ALL SELECT 23, 0, 'B'
    UNION ALL SELECT 24, 0, 'C'
    UNION ALL SELECT 25, 0, 'A'
    UNION ALL SELECT 26, 0, 'C'
    UNION ALL SELECT 27, 0, 'C'
    UNION ALL SELECT 28, 0, 'A'
    UNION ALL SELECT 29, 0, 'B'
    UNION ALL SELECT 30, 0, 'B'
    UNION ALL SELECT 31, 0, 'D'
    UNION ALL SELECT 32, 0, 'F'
    UNION ALL SELECT 33, 0, 'H'
    UNION ALL SELECT 34, 0, 'C'
    UNION ALL SELECT 35, 0, 'D'
    UNION ALL SELECT 36, 0, 'E'
    UNION ALL SELECT 37, 0, 'NOT GIVEN'
    UNION ALL SELECT 38, 0, 'YES'
    UNION ALL SELECT 39, 0, 'NO'
    UNION ALL SELECT 40, 0, 'NO'
) a
    ON a.order_index = q.order_index;

UPDATE practice_content
SET question_count = 40
WHERE practice_content_id = @practice_content_id;

DROP TEMPORARY TABLE tmp_old_practice_questions;
DROP TEMPORARY TABLE tmp_practice_questions;

COMMIT;
